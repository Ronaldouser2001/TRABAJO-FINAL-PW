"""Conexión a SQL Server mediante pyodbc, sin ORM."""

import os
from contextlib import contextmanager
from pathlib import Path
from typing import Generator

import pyodbc
from dotenv import load_dotenv

# Carga el .env del backend sin depender del directorio de ejecución
ENV_FILE = Path(__file__).resolve().parent.parent / ".env"
load_dotenv(ENV_FILE)

DEFAULT_DRIVER = "ODBC Driver 18 for SQL Server"
FALLBACK_DRIVERS = (
    "ODBC Driver 18 for SQL Server",
    "ODBC Driver 17 for SQL Server",
)


def resolve_driver(configured: str | None = None) -> str:
    """Devuelve el driver configurado o un respaldo disponible en el sistema."""
    requested = (configured or os.getenv("DB_DRIVER", "")).strip()
    installed = {driver.casefold(): driver for driver in pyodbc.drivers()}

    if requested and requested.casefold() in installed:
        return installed[requested.casefold()]

    for driver in FALLBACK_DRIVERS:
        if driver.casefold() in installed:
            return installed[driver.casefold()]

    return requested or DEFAULT_DRIVER


def get_connection_string() -> str:
    """Construye la cadena ODBC con autenticación de Windows."""
    driver = resolve_driver()
    server = os.getenv("DB_SERVER", "DESKTOP-K8VT9OI").strip()
    database = os.getenv("DB_NAME", "RunaRedBD").strip()
    trusted = os.getenv("DB_TRUSTED_CONNECTION", "yes").strip()
    trust_cert = os.getenv("DB_TRUST_SERVER_CERTIFICATE", "yes").strip()

    return (
        f"DRIVER={{{driver}}};"
        f"SERVER={server};"
        f"DATABASE={database};"
        f"Trusted_Connection={trusted};"
        f"TrustServerCertificate={trust_cert};"
    )


@contextmanager
def get_db_connection() -> Generator[pyodbc.Connection, None, None]:
    """Abre una conexión a SQL Server y garantiza su cierre."""
    connection = pyodbc.connect(get_connection_string())
    try:
        yield connection
    finally:
        connection.close()


def test_connection() -> bool:
    """Comprueba que la base de datos responde a una consulta simple."""
    try:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute("SELECT 1")
            row = cursor.fetchone()
            return row is not None and int(row[0]) == 1
    except pyodbc.Error:
        return False

from typing import Any

from app.database import get_db_connection


def _user_from_row(row: Any, include_password: bool = False) -> dict[str, Any]:
    user = {
        "id": int(row.id),
        "full_name": str(row.full_name).strip(),
        "email": str(row.email).strip(),
        "role": str(row.role).strip(),
        "created_at": row.created_at,
    }
    if include_password:
        user["password_hash"] = str(row.password_hash).strip()
    return user


class UserRepository:
    """Acceso parametrizado a la tabla dbo.users."""

    def get_by_email(self, email: str) -> dict[str, Any] | None:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                """
                SELECT id, full_name, email, password_hash, role, created_at
                FROM dbo.users
                WHERE LOWER(email) = LOWER(?)
                """,
                (email.strip(),),
            )
            row = cursor.fetchone()
            return _user_from_row(row, include_password=True) if row else None

    def get_by_id(self, user_id: int) -> dict[str, Any] | None:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                """
                SELECT id, full_name, email, password_hash, role, created_at
                FROM dbo.users
                WHERE id = ?
                """,
                (user_id,),
            )
            row = cursor.fetchone()
            return _user_from_row(row, include_password=True) if row else None

    def exists_email(self, email: str) -> bool:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                "SELECT 1 FROM dbo.users WHERE LOWER(email) = LOWER(?)",
                (email.strip(),),
            )
            return cursor.fetchone() is not None

    def create(
        self,
        full_name: str,
        email: str,
        password_hash: str,
        role: str,
    ) -> dict[str, Any]:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                """
                INSERT INTO dbo.users (full_name, email, password_hash, role)
                OUTPUT INSERTED.id, INSERTED.full_name, INSERTED.email,
                       INSERTED.role, INSERTED.created_at
                VALUES (?, ?, ?, ?)
                """,
                (
                    full_name.strip(),
                    email.strip().lower(),
                    password_hash,
                    role.strip() or "estudiante",
                ),
            )
            row = cursor.fetchone()
            connection.commit()
            if row is None:
                raise RuntimeError("SQL Server no devolvió el usuario insertado.")
            return _user_from_row(row)


user_repository = UserRepository()

"""Repositorio de recursos educativos sobre SQL Server con pyodbc."""

from typing import Any

from app.database import get_db_connection

UPDATABLE_FIELDS = (
    "title",
    "description",
    "university",
    "category",
    "file_url",
)


def _row_to_dict(cursor: Any, row: Any) -> dict[str, Any]:
    columns = [
        str(column[0]).lower() for column in (cursor.description or [])
    ]
    return dict(zip(columns, row))


class ResourceRepository:
    """Acceso parametrizado a la tabla dbo.resources."""

    def list(self, university: str | None = None) -> list[dict[str, Any]]:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            if university and university.strip():
                cursor.execute(
                    """
                    SELECT id, title, description, university, category,
                           file_url, created_at
                    FROM dbo.resources
                    WHERE university = ?
                    ORDER BY id ASC
                    """,
                    (university.strip(),),
                )
            else:
                cursor.execute(
                    """
                    SELECT id, title, description, university, category,
                           file_url, created_at
                    FROM dbo.resources
                    ORDER BY id ASC
                    """
                )
            return [_row_to_dict(cursor, row) for row in cursor.fetchall()]

    def get(self, resource_id: int) -> dict[str, Any] | None:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                """
                SELECT id, title, description, university, category,
                       file_url, created_at
                FROM dbo.resources
                WHERE id = ?
                """,
                (resource_id,),
            )
            row = cursor.fetchone()
            return _row_to_dict(cursor, row) if row else None

    def create(
        self,
        title: str,
        description: str,
        university: str,
        category: str,
        file_url: str | None = None,
    ) -> dict[str, Any]:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                """
                INSERT INTO dbo.resources
                    (title, description, university, category, file_url)
                OUTPUT INSERTED.*
                VALUES (?, ?, ?, ?, ?)
                """,
                (title, description, university, category, file_url),
            )
            row = cursor.fetchone()
            connection.commit()
            if row is None:
                raise RuntimeError("SQL Server no devolvió el recurso insertado.")
            return _row_to_dict(cursor, row)

    def update(
        self,
        resource_id: int,
        changes: dict[str, Any],
    ) -> dict[str, Any] | None:
        fields = [
            field for field in UPDATABLE_FIELDS if field in changes
        ]
        if not fields:
            return self.get(resource_id)

        assignments = ", ".join(f"{field} = ?" for field in fields)
        values = [changes[field] for field in fields]
        values.append(resource_id)

        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                f"""
                UPDATE dbo.resources
                SET {assignments}
                OUTPUT INSERTED.*
                WHERE id = ?
                """,
                values,
            )
            row = cursor.fetchone()
            connection.commit()
            return _row_to_dict(cursor, row) if row else None

    def delete(self, resource_id: int) -> bool:
        with get_db_connection() as connection:
            cursor = connection.cursor()
            cursor.execute(
                "DELETE FROM dbo.resources WHERE id = ?",
                (resource_id,),
            )
            deleted = cursor.rowcount > 0
            connection.commit()
            return deleted

"""Seguridad de contraseñas con argon2-cffi y tokens JWT con PyJWT."""

import os
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Any

import jwt
from argon2 import PasswordHasher
from argon2.exceptions import VerificationError, VerifyMismatchError
from dotenv import load_dotenv

ENV_FILE = Path(__file__).resolve().parent.parent.parent / ".env"
load_dotenv(ENV_FILE)

_password_hasher = PasswordHasher()

JWT_SECRET_KEY = os.getenv(
    "JWT_SECRET_KEY",
    "clave_super_secreta_runared_jwt_2026_al_menos_32_bytes",
)
JWT_ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "1440")
)


def hash_password(password: str) -> str:
    """Genera un hash Argon2id para la contraseña."""
    return _password_hasher.hash(password)


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Comprueba una contraseña contra su hash Argon2."""
    try:
        return _password_hasher.verify(hashed_password, plain_password)
    except (VerifyMismatchError, VerificationError):
        return False
    except Exception:
        return False


def create_access_token(
    data: dict[str, Any],
    expires_delta: timedelta | None = None,
) -> str:
    """Firma un JWT incluyendo la fecha de expiración."""
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + (
        expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    )
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, JWT_SECRET_KEY, algorithm=JWT_ALGORITHM)


def decode_access_token(token: str) -> dict[str, Any]:
    """Decodifica y valida un JWT; lanza jwt.PyJWTError si no es válido."""
    return jwt.decode(token, JWT_SECRET_KEY, algorithms=[JWT_ALGORITHM])

"""Servicio de autenticación y autorización con Argon2 y JWT."""

from typing import Annotated, Any

import jwt
from fastapi import Depends
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer

from app.repositories.user_repository import (
    UserRepository,
    user_repository,
)
from app.schemas.user import (
    Token,
    UserCreate,
    UserLogin,
    UserResponse,
)
from app.security.jwt import (
    create_access_token,
    decode_access_token,
    hash_password,
    verify_password,
)
from app.services.exceptions import (
    CredencialesInvalidasError,
    TokenInvalidoError,
    UsuarioAlreadyExistsError,
)

security_scheme = HTTPBearer(auto_error=False)


class AuthService:
    """Reglas de negocio para registro, login y usuario actual."""

    def __init__(self, repository: UserRepository | None = None) -> None:
        self._repository = repository or user_repository

    def registrar(self, datos: UserCreate) -> UserResponse:
        email = datos.email.lower().strip()
        if self._repository.exists_email(email):
            raise UsuarioAlreadyExistsError(email)

        created = self._repository.create(
            full_name=datos.full_name.strip(),
            email=email,
            password_hash=hash_password(datos.password),
            role=datos.role.strip() or "estudiante",
        )
        return self._to_response(created)

    def autenticar(self, datos: UserLogin) -> Token:
        email = datos.email.lower().strip()
        user = self._repository.get_by_email(email)
        if user is None or not verify_password(
            datos.password,
            user["password_hash"],
        ):
            raise CredencialesInvalidasError()

        access_token = create_access_token(
            data={
                "sub": str(user["id"]),
                "email": user["email"],
            }
        )
        return Token(
            access_token=access_token,
            token_type="bearer",
            user=self._to_response(user),
        )

    def obtener_usuario_actual(self, token: str) -> UserResponse:
        try:
            payload = decode_access_token(token)
            subject = payload.get("sub")
            if subject is None:
                raise TokenInvalidoError(
                    "Token inválido: identificador ausente."
                )
            user_id = int(subject)
        except (jwt.PyJWTError, TypeError, ValueError) as exc:
            raise TokenInvalidoError(
                "Token no válido o expirado."
            ) from exc

        user = self._repository.get_by_id(user_id)
        if user is None:
            raise TokenInvalidoError(
                "El usuario asociado al token ya no existe."
            )
        return self._to_response(user)

    @staticmethod
    def _to_response(user: dict[str, Any]) -> UserResponse:
        return UserResponse(
            id=int(user["id"]),
            full_name=str(user["full_name"]),
            email=str(user["email"]),
            role=str(user["role"]),
            created_at=user["created_at"],
        )


auth_service = AuthService()


def get_auth_service() -> AuthService:
    return auth_service


AuthServiceDep = Annotated[AuthService, Depends(get_auth_service)]


def get_current_user(
    credentials: Annotated[
        HTTPAuthorizationCredentials | None,
        Depends(security_scheme),
    ],
    service: AuthServiceDep,
) -> UserResponse:
    if credentials is None or not credentials.credentials:
        raise TokenInvalidoError(
            "Cabecera de autenticación Bearer requerida."
        )
    return service.obtener_usuario_actual(credentials.credentials)


CurrentUserDep = Annotated[UserResponse, Depends(get_current_user)]

"""Servicio del CRUD de recursos educativos."""

from typing import Annotated, Any

from fastapi import Depends

from app.repositories.resource_repository import (
    ResourceRepository,
    resource_repository,
)
from app.schemas.resource import (
    ResourceCreate,
    ResourceResponse,
    ResourceUpdate,
)
from app.services.exceptions import ResourceNotFoundError


class ResourceService:
    """Reglas de negocio del catálogo de recursos."""

    def __init__(
        self,
        repository: ResourceRepository | None = None,
    ) -> None:
        self._repository = repository or resource_repository

    def listar(self, university: str | None = None) -> list[ResourceResponse]:
        return [
            self._to_response(row)
            for row in self._repository.list(university)
        ]

    def obtener(self, resource_id: int) -> ResourceResponse:
        row = self._repository.get(resource_id)
        if row is None:
            raise ResourceNotFoundError(resource_id)
        return self._to_response(row)

    def crear(self, datos: ResourceCreate) -> ResourceResponse:
        row = self._repository.create(**datos.model_dump())
        return self._to_response(row)

    def actualizar(
        self,
        resource_id: int,
        datos: ResourceUpdate,
    ) -> ResourceResponse:
        changes = datos.model_dump(exclude_unset=True)
        row = self._repository.update(resource_id, changes)
        if row is None:
            raise ResourceNotFoundError(resource_id)
        return self._to_response(row)

    def eliminar(self, resource_id: int) -> None:
        if not self._repository.delete(resource_id):
            raise ResourceNotFoundError(resource_id)

    @staticmethod
    def _to_response(row: dict[str, Any]) -> ResourceResponse:
        return ResourceResponse(
            id=int(row["id"]),
            title=str(row["title"]),
            description=str(row["description"]),
            university=str(row["university"]),
            category=str(row["category"]),
            file_url=row["file_url"],
            created_at=row["created_at"],
        )


resource_service = ResourceService()


def get_resource_service() -> ResourceService:
    return resource_service


ResourceServiceDep = Annotated[
    ResourceService,
    Depends(get_resource_service),
