  FormBuilder,

  Validators
} from '@angular/forms';
import { RouterLink } from '@angular/router';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [ReactiveFormsModule, RouterLink],
  templateUrl: './login.html',
  styleUrl: './login.css'
})
export class Login {
  private fb = inject(FormBuilder);

  mostrarClave = signal(false);
  mensaje = signal('');

  formulario = this.fb.nonNullable.group({
    correo: ['', [Validators.required, Validators.email]],
    clave: ['', Validators.required]

  get correo() {
    return this.formulario.controls.correo;
  }

  get clave() {

  }

    this.mensaje.set('');
    this.formulario.markAllAsTouched();

import { Component, computed, inject, OnInit, signal } from '@angular/core';
import { ResourceService, ResourceResponse } from '../../services/resource.service';

@Component({
  selector: 'app-recursos',
  standalone: true,
  imports: [],
  templateUrl: './recursos.html',
  styleUrl: './recursos.css'
})
export class Recursos implements OnInit {
  private resourceService = inject(ResourceService);

  busqueda = signal<string>('');
  universidadFiltro = signal<string>('Todas');
  recursos = signal<ResourceResponse[]>([]);
  cargando = signal<boolean>(true);
  errorMsg = signal<string>('');

  universidades: string[] = ['Todas', 'UNCP', 'UPLA', 'UC', 'Roosevelt'];

  ngOnInit(): void {
    this.cargarRecursos();
  }

  cargarRecursos(): void {
    this.cargando.set(true);
    this.errorMsg.set('');

    this.resourceService.getResources().subscribe({
      next: (data: ResourceResponse[]) => {
        this.recursos.set(data);
        this.cargando.set(false);
      },
      error: (err: unknown) => {
        console.error('Error al conectar con la API:', err);
        this.errorMsg.set('No se pudo conectar con el servidor de RunaRed.');
        this.cargando.set(false);
      }
    });
  }

  recursosFiltrados = computed(() => {
    const texto = this.normalizar(this.busqueda().trim());
    const univ = this.universidadFiltro();

    return this.recursos().filter((recurso) => {
      const contenido = this.normalizar(
        `${recurso.title} ${recurso.category} ${recurso.description} ${recurso.university}`
      );

      const coincideTexto = contenido.includes(texto);
      const coincideUniversidad = univ === 'Todas' || recurso.university === univ;

      return coincideTexto && coincideUniversidad;
    });
  });

  limpiarFiltros(): void {
    this.busqueda.set('');
    this.universidadFiltro.set('Todas');
  }

  private normalizar(texto: string): string {
    return texto
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '');
  }
}

import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface ResourceResponse {
  id: number;
  title: string;
  description: string;
  university: string;
  category: string;
  file_url?: string;
  created_at: string;
}

export interface ResourceCreate {
  title: string;
  description: string;
  university: string;
  category: string;
  file_url?: string;
}

@Injectable({
  providedIn: 'root'
})
export class ResourceService {
  private http = inject(HttpClient);
  private apiUrl = 'http://127.0.0.1:8000/resources';

  getResources(university?: string): Observable<ResourceResponse[]> {
    const url = university ? `${this.apiUrl}/?university=${university}` : `${this.apiUrl}/`;
    return this.http.get<ResourceResponse[]>(url);
  }

  createResource(resource: ResourceCreate): Observable<ResourceResponse> {
    return this.http.post<ResourceResponse>(`${this.apiUrl}/`, resource);
  }

  getResourceById(id: number): Observable<ResourceResponse> {
    return this.http.get<ResourceResponse>(`${this.apiUrl}/${id}`);
  }

  deleteResource(id: number): Observable<void> {
    return this.http.delete<void>(`${this.apiUrl}/${id}`);
  }
}
