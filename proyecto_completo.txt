from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from app.database import get_db
from app.models.user import User
from app.schemas.user import UserCreate, UserLogin, UserResponse, Token
from app.security.jwt import hash_password, verify_password, create_access_token, get_current_user

router = APIRouter(prefix="/auth", tags=["AutenticaciÃ³n"])

@router.post("/registro", response_model=UserResponse, status_code=status.HTTP_201_CREATED)
def register(user_data: UserCreate, db: Session = Depends(get_db)):
    existing = db.query(User).filter(User.email == user_data.email).first()
    if existing:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="El correo electrÃ³nico ya estÃ¡ registrado"
        )
    new_user = User(
        full_name=user_data.full_name.strip(),
        email=user_data.email.lower().strip(),
        password_hash=hash_password(user_data.password),
        role=user_data.role
    )
    db.add(new_user)
    db.commit()
    db.refresh(new_user)
    return new_user

@router.post("/login", response_model=Token)
def login(credentials: UserLogin, db: Session = Depends(get_db)):
    user = db.query(User).filter(User.email == credentials.email.lower().strip()).first()
    if not user or not verify_password(credentials.password, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Correo o contraseÃ±a incorrectos"
        )
    token = create_access_token(data={"sub": user.email, "role": user.role})
    return {"access_token": token, "token_type": "bearer", "user": user}

@router.get("/me", response_model=UserResponse)
def get_me(current_user: User = Depends(get_current_user)):
    return current_user

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from typing import List, Optional
from app.database import get_db
from app.models.resource import Resource
from app.schemas.resource import ResourceCreate, ResourceUpdate, ResourceResponse

router = APIRouter(prefix="/resources", tags=["Recursos Educativos"])

# 1. CREATE: Crear un nuevo recurso
@router.post("/", response_model=ResourceResponse, status_code=status.HTTP_201_CREATED)
def create_resource(resource: ResourceCreate, db: Session = Depends(get_db)):
    new_resource = Resource(**resource.model_dump())
    db.add(new_resource)
    db.commit()
    db.refresh(new_resource)
    return new_resource

# 2. READ ALL: Obtener todos los recursos
@router.get("/", response_model=List[ResourceResponse])
def get_resources(university: Optional[str] = None, db: Session = Depends(get_db)):
    query = db.query(Resource)
    if university:
        query = query.filter(Resource.university == university)
    return query.all()

# 3. READ ONE: Obtener un solo recurso por ID
@router.get("/{resource_id}", response_model=ResourceResponse)
def get_resource(resource_id: int, db: Session = Depends(get_db)):
    resource = db.query(Resource).filter(Resource.id == resource_id).first()
    if not resource:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail=f"Recurso con ID {resource_id} no encontrado"
        )
    return resource

# 4. UPDATE: Actualizar un recurso por ID
@router.put("/{resource_id}", response_model=ResourceResponse)
def update_resource(resource_id: int, updated_data: ResourceUpdate, db: Session = Depends(get_db)):
    query = db.query(Resource).filter(Resource.id == resource_id)
    resource = query.first()
    
    if not resource:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail=f"Recurso con ID {resource_id} no encontrado"
        )
    
    update_dict = updated_data.model_dump(exclude_unset=True)
    query.update(update_dict, synchronize_session=False)
    db.commit()
    db.refresh(resource)
    return resource

# 5. DELETE: Eliminar un recurso por ID
@router.delete("/{resource_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_resource(resource_id: int, db: Session = Depends(get_db)):
    resource = db.query(Resource).filter(Resource.id == resource_id).first()
    if not resource:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail=f"Recurso con ID {resource_id} no encontrado"
        )
    
    db.delete(resource)
    db.commit()
    return None

import os
from datetime import datetime, timedelta, timezone
from typing import Optional
import bcrypt
from jose import JWTError, jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.orm import Session
from app.database import get_db
from app.models.user import User

SECRET_KEY = os.getenv("JWT_SECRET_KEY", "cambia-esta-clave-secreta-en-produccion-runared-2026")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 60 * 24  # 24 horas

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/auth/login")

def hash_password(password: str) -> str:
    salt = bcrypt.gensalt()
    return bcrypt.hashpw(password.encode("utf-8"), salt).decode("utf-8")

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return bcrypt.checkpw(plain_password.encode("utf-8"), hashed_password.encode("utf-8"))

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + (expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def get_current_user(token: str = Depends(oauth2_scheme), db: Session = Depends(get_db)) -> User:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="No se pudieron validar las credenciales",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        email: str = payload.get("sub")
        if email is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception

    user = db.query(User).filter(User.email == email).first()
    if user is None:
        raise credentials_exception
    return user

export const authGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.isAuthenticated() && !auth.tokenVencido()) return true;

  auth.limpiarSesion();
  return router.createUrlTree(["/login"]);
};

export const guestGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (!auth.isAuthenticated() || auth.tokenVencido()) {
    if (auth.tokenVencido()) auth.limpiarSesion();
    return true;
  }

  return router.createUrlTree(["/recursos"]);
};

import {
  HttpErrorResponse,
  HttpInterceptorFn,
} from "@angular/common/http";
import { inject } from "@angular/core";
import { catchError, throwError } from "rxjs";
import { AuthService } from "../services/auth.service";

function esRutaDeAutenticacion(url: string): boolean {
  return /\/auth\/(login|registro)(?:[?#]|$)/.test(url);
}

export const authInterceptor: HttpInterceptorFn = (request, next) => {
  const auth = inject(AuthService);
  const token = auth.token();
  const esAuth = esRutaDeAutenticacion(request.url);
  const peticion =
    token && !esAuth
      ? request.clone({
          setHeaders: { Authorization: `Bearer ${token}` },
        })
      : request;

  return next(peticion).pipe(
    catchError((error: unknown) => {
      // Un 401 en una ruta protegida invalida la sesiÃ³n (token vencido o revocado).
      if (
        error instanceof HttpErrorResponse &&
        error.status === 401 &&
        !esAuth &&
        token
      ) {
        auth.logout();
      }
      return throwError(() => error);
    }),
  );
};

@Component({
  selector: "app-login",
  standalone: true,
  imports: [ReactiveFormsModule, RouterLink],
  templateUrl: "./login.html",
})
export class Login {
  private auth = inject(AuthService);
  private router = inject(Router);
  private fb = inject(FormBuilder);
  cargando = signal(false);
  error = signal("");
  form = this.fb.nonNullable.group({
    email: ["", [Validators.required, Validators.email]],
    password: ["", [Validators.required]],
  });

  invalido(campo: "email" | "password") {
    const control = this.form.controls[campo];
    return control.touched && control.invalid;
  }

  enviar() {
    if (this.cargando()) return;
    this.form.markAllAsTouched();
    this.error.set("");
    if (this.form.invalid) return;

    this.cargando.set(true);
    this.auth
      .login(this.form.getRawValue())
      .pipe(finalize(() => this.cargando.set(false)))
      .subscribe({
        next: () => void this.router.navigateByUrl("/recursos"),
        error: (e) => this.error.set(apiError(e)),
      });
  }
}

@Component({
  selector: "app-registro",
  standalone: true,
  imports: [ReactiveFormsModule, RouterLink],
  templateUrl: "./registro.html",
})
export class Registro {
  private auth = inject(AuthService);
  private router = inject(Router);
  private fb = inject(FormBuilder);
  cargando = signal(false);
  error = signal("");
  form = this.fb.nonNullable.group(
    {
      full_name: ["", [Validators.required, Validators.maxLength(120)]],
      email: ["", [Validators.required, Validators.email]],
      password: ["", [Validators.required, Validators.minLength(8)]],
      confirmar: ["", [Validators.required]],
      role: ["estudiante", [Validators.required]],
    },
    {
      validators: (group) =>
        group.get("password")?.value === group.get("confirmar")?.value
          ? null
          : { noCoincide: true },
    },
  );

  invalido(campo: "full_name" | "email" | "password" | "confirmar") {
    const control = this.form.controls[campo];
    return control.touched && control.invalid;
  }

  noCoincide() {
    return (
      this.form.hasError("noCoincide") &&
      (this.form.controls.password.touched ||
        this.form.controls.confirmar.touched)
    );
  }

  enviar() {
    if (this.cargando()) return;
    this.form.markAllAsTouched();
    this.error.set("");
    if (this.form.invalid) return;

    const { full_name, email, password, role } = this.form.getRawValue();
    this.cargando.set(true);
    this.auth
      .register({ full_name: full_name.trim(), email, password, role })
      .pipe(finalize(() => this.cargando.set(false)))
      .subscribe({
        next: () => void this.router.navigateByUrl("/recursos"),
        error: (e) => this.error.set(apiError(e)),
      });
  }
}

import { Component, computed, inject, signal } from "@angular/core";
import { FormControl, ReactiveFormsModule } from "@angular/forms";
import { ActivatedRoute, RouterLink } from "@angular/router";
import { toSignal } from "@angular/core/rxjs-interop";
import { catchError, map, of, startWith, Subject, switchMap } from "rxjs";
import {
  ResourceResponse,
  ResourceService,
} from "../../services/resource.service";
import { apiError } from "../../core/api-error";
interface Estado {
  cargando: boolean;
  error: string;
  recursos: ResourceResponse[];
}
@Component({
  selector: "app-catalogo-api",
  standalone: true,
  imports: [RouterLink, ReactiveFormsModule],
  templateUrl: "./catalogo.html",
})
export class Catalogo {
  eliminado =
    inject(ActivatedRoute).snapshot.queryParamMap.get("actualizado") ===
    "eliminado";
  private api = inject(ResourceService);
  private recargar = new Subject<void>();
  busqueda = signal("");
  categoria = signal("Todas");
  pagina = signal(1);
  universidad = new FormControl("", { nonNullable: true });
  estado = toSignal(
    this.recargar.pipe(
      startWith(undefined),
      switchMap(() =>
        this.api.getResources(this.universidad.value).pipe(
          map((recursos) => ({ cargando: false, error: "", recursos })),
          catchError((e) =>
            of({ cargando: false, error: apiError(e), recursos: [] }),
          ),
          startWith({ cargando: true, error: "", recursos: [] } as Estado),
        ),
      ),
    ),
    { initialValue: { cargando: true, error: "", recursos: [] } as Estado },
  );
  categorias = computed(() => [
    "Todas",
    ...new Set(this.estado().recursos.map((r) => r.category)),
  ]);
  filtrados = computed(() =>
    this.estado().recursos.filter(
      (r) =>
        this.normalizar(
          r.title + " " + r.description + " " + r.university,
        ).includes(this.normalizar(this.busqueda())) &&
        (this.categoria() === "Todas" || r.category === this.categoria()),
    ),
  );
  totalPaginas = computed(() =>
    Math.max(1, Math.ceil(this.filtrados().length / 9)),
  );
  visibles = computed(() =>
    this.filtrados().slice((this.pagina() - 1) * 9, this.pagina() * 9),
  );
  normalizar(s: string) {
    return s
      .toLowerCase()
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g, "")
      .trim();
  }
  cargar() {
    this.pagina.set(1);
    this.categoria.set("Todas");
    this.recargar.next();
  }
  buscar(s: string) {
    this.busqueda.set(s);
    this.pagina.set(1);
  }
  tipo(s: string) {
    this.categoria.set(s);
    this.pagina.set(1);
  }
  limpiar() {
    this.busqueda.set("");
    this.universidad.setValue("");
    this.cargar();
  }
}

import { HttpClient } from "@angular/common/http";
import { computed, inject, Injectable, signal } from "@angular/core";
import { map, Observable, switchMap, tap, timeout } from "rxjs";
import { environment } from "../../environments/environment";

export interface AuthUser {
  id: number;
  full_name: string;
  email: string;
  role: string;
  created_at: string;
}

export interface LoginRequest {
  email: string;
  password: string;
}

export interface RegisterRequest extends LoginRequest {
  full_name: string;
  role: string;
}

export interface AuthSession {
  access_token: string;
  token_type: string;
  user: AuthUser;
}

const TOKEN_KEY = "runared_token";
const USER_KEY = "runared_user";

function isAuthUser(value: unknown): value is AuthUser {
  if (!value || typeof value !== "object") return false;
  const user = value as Record<string, unknown>;
  return (
    Number.isSafeInteger(user["id"]) &&
    Number(user["id"]) > 0 &&
    typeof user["full_name"] === "string" &&
    typeof user["email"] === "string" &&
    typeof user["role"] === "string" &&
    typeof user["created_at"] === "string"
  );
}

function parseSession(value: unknown): AuthSession {
  if (!value || typeof value !== "object")
    throw new Error("Respuesta de autenticaciÃ³n invÃ¡lida.");
  const session = value as Record<string, unknown>;
  if (
    typeof session["access_token"] !== "string" ||
    !session["access_token"].trim() ||
    typeof session["token_type"] !== "string" ||
    !isAuthUser(session["user"])
  )
    throw new Error("La respuesta de autenticaciÃ³n no coincide con el contrato.");
  return {
    access_token: session["access_token"],
    token_type: session["token_type"],
    user: session["user"],
  };
}

function readToken(): string | null {
  try {
    return localStorage.getItem(TOKEN_KEY);
  } catch {
    return null;
  }
}

function readUser(): AuthUser | null {
  try {
    const raw = localStorage.getItem(USER_KEY);
    if (!raw) return null;
    const value: unknown = JSON.parse(raw);
    return isAuthUser(value) ? value : null;
  } catch {
    return null;
  }
}

function expiration(token: string): number | null {
  try {
    const payload = token.split(".")[1];
    if (!payload) return null;
    const normalized = payload.replace(/-/g, "+").replace(/_/g, "/");
    const padded = normalized.padEnd(
      Math.ceil(normalized.length / 4) * 4,
      "=",
    );
    const claims: unknown = JSON.parse(atob(padded));
    if (!claims || typeof claims !== "object") return null;
    const exp = (claims as Record<string, unknown>)["exp"];
    return typeof exp === "number" ? exp * 1000 : null;
  } catch {
    return null;
  }
}

@Injectable({ providedIn: "root" })
export class AuthService {
  private http = inject(HttpClient);
  private readonly base =
    environment.apiBaseUrl.replace(/\/$/, "") + "/auth";
  private readonly _token = signal<string | null>(readToken());
  private readonly _currentUser = signal<AuthUser | null>(readUser());

  readonly token = this._token.asReadonly();
  readonly currentUser = this._currentUser.asReadonly();
  readonly isAuthenticated = computed(
    () => this._token() !== null && this._currentUser() !== null,
  );

  constructor() {
    if (this.tokenVencido()) this.limpiarSesion();
  }

  login(credentials: LoginRequest): Observable<AuthSession> {
    return this.http
      .post<unknown>(this.base + "/login", credentials)
      .pipe(
        timeout(15000),
        map(parseSession),
        tap((session) => this.guardarSesion(session)),
      );
  }

  register(data: RegisterRequest): Observable<AuthSession> {
    return this.http
      .post<unknown>(this.base + "/registro", data)
      .pipe(
        timeout(15000),
        switchMap(() =>
          this.login({ email: data.email, password: data.password }),
        ),
      );
  }

  logout(): void {
    this.limpiarSesion();
  }

  limpiarSesion(): void {
    this._token.set(null);
    this._currentUser.set(null);
    try {
      localStorage.removeItem(TOKEN_KEY);
      localStorage.removeItem(USER_KEY);
    } catch {
      // Almacenamiento no disponible: la sesiÃ³n en memoria ya fue limpiada.
    }
  }

  tokenVencido(): boolean {
    const token = this._token();
    if (!token) return false;
    const expiresAt = expiration(token);
    return expiresAt !== null && expiresAt <= Date.now();
  }

  private guardarSesion(session: AuthSession): void {
    this._token.set(session.access_token);
    this._currentUser.set(session.user);
    try {
      localStorage.setItem(TOKEN_KEY, session.access_token);
      localStorage.setItem(USER_KEY, JSON.stringify(session.user));
    } catch {
      // La sesiÃ³n sigue activa en memoria aunque el almacenamiento falle.
    }
  }
}

import { inject, Injectable } from "@angular/core";
import { HttpClient, HttpParams } from "@angular/common/http";
import { map, Observable, timeout } from "rxjs";
import { environment } from "../../environments/environment";
export interface ResourceCreate {
  title: string;
  description: string;
  university: string;
  category: string;
  file_url: string | null;
}
export interface ResourceResponse extends ResourceCreate {
  id: number;
  created_at: string;
}
export type ResourceUpdate = Partial<ResourceCreate>;
export function safeResourceUrl(raw: string | null | undefined): string | null {
  if (!raw?.trim()) return null;
  try {
    const url = new URL(raw.trim());
    return ["https:", "http:"].includes(url.protocol) &&
      !url.username &&
      !url.password
      ? url.href
      : null;
  } catch {
    return null;
  }
}
function resource(value: unknown): ResourceResponse {
  if (!value || typeof value !== "object")
    throw new Error("Respuesta de recurso invÃ¡lida.");
  const r = value as Record<string, unknown>;
  if (
    !Number.isInteger(r["id"]) ||
    Number(r["id"]) <= 0 ||
    ["title", "description", "university", "category", "created_at"].some(
      (k) => typeof r[k] !== "string",
    ) ||
    (r["file_url"] != null && typeof r["file_url"] !== "string")
  )
    throw new Error(
      "La respuesta de la API no coincide con el contrato de recursos.",
    );
  return {
    id: Number(r["id"]),
    title: r["title"] as string,
    description: r["description"] as string,
    university: r["university"] as string,
    category: r["category"] as string,
    file_url: (r["file_url"] as string | null) ?? null,
    created_at: r["created_at"] as string,
  };
}
@Injectable({ providedIn: "root" })
export class ResourceService {
  private http = inject(HttpClient);
  private readonly url =
    environment.apiBaseUrl.replace(/\/$/, "") + "/resources";
  getResources(university?: string): Observable<ResourceResponse[]> {
    let params = new HttpParams();
    if (university?.trim())
      params = params.set("university", university.trim());
    return this.http.get<unknown>(this.url + "/", { params }).pipe(
      timeout(15000),
      map((data) => {
        if (!Array.isArray(data))
          throw new Error("Se esperaba una lista de recursos.");
        return data.map(resource);
      }),
    );
  }
  getResourceById(id: number): Observable<ResourceResponse> {
    return this.http
      .get<unknown>(this.url + "/" + id)
      .pipe(timeout(15000), map(resource));
  }
  createResource(body: ResourceCreate): Observable<ResourceResponse> {
    return this.http
      .post<unknown>(this.url + "/", body)
      .pipe(timeout(15000), map(resource));
  }
  updateResource(
    id: number,
    body: ResourceUpdate,
  ): Observable<ResourceResponse> {
    return this.http
      .put<unknown>(this.url + "/" + id, body)
      .pipe(timeout(15000), map(resource));
  }
  deleteResource(id: number): Observable<void> {
    return this.http.delete<void>(this.url + "/" + id).pipe(timeout(15000));
  }
}
